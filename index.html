function speak(text) { if (!("speechSynthesis" in window)) { alert("Voice support उपलब्ध नहीं है"); return; } speechSynthesis.cancel(); const msg = new SpeechSynthesisUtterance(text); const voices = speechSynthesis.getVoices(); const hindiVoices = voices.filter(v => v.lang.toLowerCase().startsWith("hi") ); const femaleVoice = hindiVoices.find(v => /female|woman|swara|heera|kalpana/i.test(v.name) ); if (femaleVoice) { msg.voice = femaleVoice; } else if (hindiVoices.length > 0) { msg.voice = hindiVoices[0]; } msg.lang = "hi-IN"; msg.rate = 0.88; msg.pitch = 1.08; msg.volume = 1; speechSynthesis.speak(msg); }
